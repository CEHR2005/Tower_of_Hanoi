cmake_minimum_required(VERSION 3.20)
project(hanoi VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(hanoi_core INTERFACE)
target_include_directories(hanoi_core INTERFACE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(hanoi_core INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_library(hanoi_visual src/visualization.cpp)
target_include_directories(hanoi_visual PUBLIC src)
target_link_libraries(hanoi_visual PUBLIC hanoi_core)
find_package(Threads REQUIRED)
add_executable(hanoi src/main.cpp)
target_link_libraries(hanoi PRIVATE hanoi_visual Threads::Threads)
add_executable(hanoi_bench benchmarks/bench.cpp)
target_link_libraries(hanoi_bench PRIVATE hanoi_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(hanoi_tests tests/core.cpp)
    target_link_libraries(hanoi_tests PRIVATE hanoi_visual)
    add_test(NAME core COMMAND hanoi_tests)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:hanoi>)
endif()
